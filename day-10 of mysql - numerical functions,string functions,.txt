CHECK IF ANY EMPLOYEES FIRST_NAME IS A PALINDROME

SELECT FIRST_NAME,
CONCAT(LEFT(EMAIL,3),'***@',SUBSTRING_INDEX(EMAIL,'@',-1))AS MASKED_EMAIL FROM EMPLOYEES;

SELECT first_name
FROM employees
WHERE LOWER(first_name) = REVERSE(LOWER(first_name));

SELECT FIRST_NAME,
CASE WHEN LENGTH(FIRST_NAME)<=5 THEN 'SHORT NAME'
ELSE 'LONG NAME' END AS NAME_CATEGORY
FROM EMPLOYEES;
